    payload: Dict[str, Any] = field(default_factory=dict)
    app_id: Optional[str] = None
    level: str = "info"
    ts: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())

# Module-level reference to the store (set by store.py on init to avoid circular imports)
_store: Optional[Any] = None


def _set_store(store: Any) -> None:
    """Called by SQLiteStore.__init__ to register itself."""
    global _store
    _store = store


def emit(
    agent: str,
    app_id: Optional[str],
    type: str,
    payload: Optional[Dict[str, Any]] = None,
    level: str = "info",
) -> Event:
    """Create an Event, persist it (if store is ready), and print JSON to stdout."""
    ev = Event(
        agent=agent,
        app_id=app_id,
        type=type,
        payload=payload or {},
        level=level,
    )
    if _store is not None:
        try:
            _store.emit_event(ev)
        except Exception:
            pass  # never crash an agent because of event logging
    else:
        _print_event(ev)
    return ev


def _print_event(ev: Event) -> None:
    d = asdict(ev)
    d.pop("id", None)
    print(json.dumps(d), flush=True)


def events_since(last_id: int) -> list[Event]:
    """Return events with id > last_id from the store (empty list if no store)."""
    if _store is None:
        return []
    return _store.list_events(since_id=last_id)

class SQLiteStore:
    def __init__(self, db_path: str = "state.db") -> None:
        self._db_path = db_path
        self._local = threading.local()
        self._init_db()
        # Register ourselves so events.py can call emit_event()
        _events_mod._set_store(self)

    # ── Connection management ───────────────────────────────────────────────
    def _conn(self) -> sqlite3.Connection:
        if not hasattr(self._local, "conn") or self._local.conn is None:
            self._local.conn = sqlite3.connect(self._db_path, check_same_thread=False)
            self._local.conn.row_factory = sqlite3.Row
        return self._local.conn

    @contextmanager
    def _cursor(self):
        conn = self._conn()
        cur = conn.cursor()
        try:
            yield cur
            conn.commit()
        except Exception:
            conn.rollback()
            raise
        finally:
            cur.close()

    def _init_db(self) -> None:
        conn = self._conn()
        conn.executescript(_CREATE_SQL)
        conn.commit()

    # ── Apps ───────────────────────────────────────────────────────────────
    def upsert_apps(self, apps: list[AppRecord]) -> None:
        with self._cursor() as cur:
            for app in apps:
                cur.execute(
                    "INSERT OR REPLACE INTO apps(app_id, source, status, json) VALUES (?,?,?,?)",
                    (app.app_id, app.source, app.status, app.model_dump_json()),
                )

    def upsert_app(self, app: AppRecord) -> None:
        self.upsert_apps([app])

    def get_app(self, app_id: str) -> Optional[AppRecord]:
        with self._cursor() as cur:
            row = cur.execute("SELECT json FROM apps WHERE app_id=?", (app_id,)).fetchone()
        return AppRecord.model_validate_json(row["json"]) if row else None

    def get_apps(
        self,
        source: Optional[str] = None,
        tier: Optional[str] = None,
        status: Optional[str] = None,
    ) -> list[AppRecord]:
        query = "SELECT a.json FROM apps a"
        conditions: list[str] = []
        params: list = []
        if source:
            conditions.append("a.source=?")
            params.append(source)
        if status:
            conditions.append("a.status=?")
            params.append(status)
        if tier:
            query += " JOIN tiers t ON a.app_id=t.app_id"
            conditions.append("t.tier=?")
            params.append(tier)
        if conditions:
            query += " WHERE " + " AND ".join(conditions)
        with self._cursor() as cur:
            rows = cur.execute(query, params).fetchall()
        return [AppRecord.model_validate_json(r["json"]) for r in rows]

    # Keep old name for compatibility
    def list_apps(
        self,
        source: Optional[str] = None,
        tier: Optional[str] = None,
        status: Optional[str] = None,
    ) -> list[AppRecord]:
        return self.get_apps(source=source, tier=tier, status=status)

    def set_status(self, app_id: str, status: str) -> None:
        app = self.get_app(app_id)
        if app:
            app.status = status  # type: ignore[assignment]
            self.upsert_app(app)

    # ── Tiers ──────────────────────────────────────────────────────────────
    def upsert_tiers(self, tiers: list[Tiering]) -> None:
        with self._cursor() as cur:
            for t in tiers:
                cur.execute(
                    "INSERT OR REPLACE INTO tiers(app_id, tier, json) VALUES (?,?,?)",
                    (t.app_id, t.tier, t.model_dump_json()),
                )

    def upsert_tiering(self, t: Tiering) -> None:
        self.upsert_tiers([t])

    def get_tiers(self) -> list[Tiering]:
        with self._cursor() as cur:
            rows = cur.execute("SELECT json FROM tiers").fetchall()
        return [Tiering.model_validate_json(r["json"]) for r in rows]

    # ── Edges ──────────────────────────────────────────────────────────────
    def replace_edges(self, edges: list[Edge]) -> None:
        with self._cursor() as cur:
            cur.execute("DELETE FROM edges")
            for e in edges:
                cur.execute(
                    "INSERT INTO edges(from_app_id, to_app_id, signals, source) VALUES (?,?,?,?)",
                    (e.from_app_id, e.to_app_id, json.dumps(e.signals), e.source),
                )

    def get_edges(self) -> list[Edge]:
        with self._cursor() as cur:
            rows = cur.execute("SELECT * FROM edges").fetchall()
        result = []
        for r in rows:
            result.append(

        system: str,
        messages: list[dict],
        tools: list[dict],

        Run a tool-calling loop.

        Returns the final assistant message dict (with "content" list containing
        tool_use or text blocks) or None if backend is "off".

            return None

            return None
        # Return the first matching canned response for the first available tool
        tool_name = tools[0]["name"] if tools else ""
        canned = self._canned.get(tool_name, self._canned.get("default"))
        if canned is None:
            return None
        return {
            "role": "assistant",
            "content": [
                {
                    "type": "tool_use",
                    "id": "mock-tool-use-1",
                    "name": tool_name,
                    "input": canned,
                }
            ],
        }

    # ── Bedrock tool-call loop ─────────────────────────────────────────────
    def _bedrock_loop(
        self,
        system: str,
        messages: list[dict],
        tools: list[dict],
        max_turns: int,
    ) -> dict:
        import anthropic

        msgs = list(messages)
        for _ in range(max_turns):
            response = self._client.messages.create(
                model=self.model_id,
                max_tokens=4096,
                system=system,
                messages=msgs,
                tools=tools,
            )
            assistant_msg = {"role": "assistant", "content": response.content}
            msgs.append(assistant_msg)

            # Check if there are tool_use blocks
            tool_uses = [b for b in response.content if b.type == "tool_use"]
            if not tool_uses:
                # No more tool calls — return the final message
                return assistant_msg

            # Build a tool_result message so the loop can continue
            tool_results = []
            for tu in tool_uses:

                # assistant turn that contains tool_use so the caller can act.
                # For the loop to work, we'd need a dispatcher here.
                # Return the message immediately — caller handles execution.
                pass

            # Return the first assistant message with tool_use blocks;
            # the agent's run_agent() loop in loop.py handles execution.
            return assistant_msg

        return msgs[-1]

run_agent — thin LLM tool-calling loop shared by all agents.
Each step is logged as an event.
"""
from __future__ import annotations

import json
from typing import Any, Callable

from agents.common.events import emit
from agents.common.llm import LLMClient
from agents.common.store import SQLiteStore


def run_agent(
    agent_name: str,
    system: str,
    user_prompt: str,
    tools: list[dict],
    tool_dispatch: dict[str, Callable[..., Any]],
    store: SQLiteStore,
    app_id: str | None = None,
    max_turns: int = 10,
) -> list[Any]:
    """
    Run one agent's tool-calling loop.

    Parameters
    ----------
    agent_name    : used in event.agent
    system        : system prompt for the LLM
    user_prompt   : first user message
    tools         : Anthropic-format tool definitions
    tool_dispatch : {tool_name: callable(**input) -> Any}
    store         : active SQLiteStore
    app_id        : optional, forwarded to events
    max_turns     : max LLM turns

    Returns
    -------
    list of tool results in order
    """
    llm = LLMClient()
    results: list[Any] = []
    messages = [{"role": "user", "content": user_prompt}]

    emit(agent_name, app_id, "TOOL_CALL", {"step": "start", "prompt": user_prompt[:200]})

    for turn in range(max_turns):
        response = llm.call_with_tools(system, messages, tools, max_turns=1)

        if response is None:
            # LLM_BACKEND=off — rule-based path; caller should not have called us
            emit(agent_name, app_id, "LLM_FALLBACK", {"turn": turn})
            break

        messages.append(response)

        # Extract tool_use blocks
        tool_uses = [
            b for b in (response.get("content") or []) if
            (isinstance(b, dict) and b.get("type") == "tool_use") or
            (hasattr(b, "type") and b.type == "tool_use")
        ]

        if not tool_uses:
            # Final text response — done
            break

        tool_result_contents = []
        for tu in tool_uses:
            name = tu["name"] if isinstance(tu, dict) else tu.name
            inp = tu["input"] if isinstance(tu, dict) else tu.input
            tool_id = tu.get("id", "tid") if isinstance(tu, dict) else tu.id

            emit(agent_name, app_id, "TOOL_CALL", {"tool": name, "input": inp})

            if name in tool_dispatch:
                try:
                    result = tool_dispatch[name](**inp)
                    results.append(result)
                    tool_result_contents.append({
                        "type": "tool_result",
                        "tool_use_id": tool_id,
                        "content": json.dumps(result),
                    })
                    emit(agent_name, app_id, "TOOL_CALL", {"tool": name, "result": str(result)[:200]})
                except Exception as exc:
                    tool_result_contents.append({
                        "type": "tool_result",
                        "tool_use_id": tool_id,
                        "content": f"ERROR: {exc}",
                        "is_error": True,
                    })
                    emit(agent_name, app_id, "ERROR", {"tool": name, "error": str(exc)}, level="error")
            else:
                tool_result_contents.append({
                    "type": "tool_result",
                    "tool_use_id": tool_id,
                    "content": f"Unknown tool: {name}",
                    "is_error": True,
                })

        messages.append({"role": "user", "content": tool_result_contents})

    return results

# Cutover stub
# ---------------------------------------------------------------------------

def run_cutover(
    store: SQLiteStore,
    app_id: str,
    steps: tuple[int, ...] = (10, 50, 100),
    bad_wave: bool = False,
) -> CutoverRun:
    from agents.common.models import CutoverStep

    emit("cutover", app_id, "CUTOVER_STARTED", {"steps": list(steps)})
    store.set_status(app_id, "CUTTING_OVER")
    time.sleep(0.1)

    history: list[CutoverStep] = []
    rolled_back = False

    for weight in steps:
        emit("cutover", app_id, "WEIGHT_SET", {"weight": weight})
        time.sleep(0.5)

        # Simulate bad wave: first step fails with high error rate
        if bad_wave and weight == steps[0]:
            error_rate = 0.38
            p95_ms = 240.0
            gate = "FAIL"
            history.append(CutoverStep(
                weight=weight,
                ts=datetime.now(timezone.utc).isoformat(),
                error_rate=error_rate,
                p95_ms=p95_ms,
                target_share=weight / 100,
                gate=gate,
            ))
            emit("cutover", app_id, "GATE_FAIL", {
                "weight": weight,
                "error_rate": error_rate,
                "p95_ms": p95_ms,
            })
            emit("cutover", app_id, "ROLLED_BACK", {
                "at_weight": weight,
                "reason": f"error_rate {error_rate:.0%} > 2%",
                "llm_explanation": (
                    f"Rolled back at {weight}%: {error_rate:.0%} 5xx from target. "
                    "Likely a missing UPSTREAM_URL env var in the blueprint."
                ),
            })
            rolled_back = True
            store.set_status(app_id, "ROLLED_BACK")
            run = CutoverRun(
                run_id=f"cut-{app_id}-stub",
                app_id=app_id,
                source="real",
                steps=list(steps),
                history=history,
                result="ROLLED_BACK",
                rollback=RollbackInfo(at_weight=weight, reason=f"error_rate {error_rate:.0%} > 2%"),
                explanation=(
                    f"Rolled back at {weight}%: {error_rate:.0%} 5xx from target. "
                    "Likely a missing UPSTREAM_URL."
                ),
            )
            store.save_cutover(run)
            return run

        # Normal step
        error_rate = 0.0
        p95_ms = 120.0 + weight * 0.5
        history.append(CutoverStep(
            weight=weight,
            ts=datetime.now(timezone.utc).isoformat(),
            error_rate=error_rate,
            p95_ms=p95_ms,
            target_share=weight / 100,
            gate="PASS",
        ))
        emit("cutover", app_id, "GATE_PASS", {
            "weight": weight,
            "error_rate": error_rate,
            "p95_ms": p95_ms,
        })

    store.set_status(app_id, "MIGRATED")
    emit("cutover", app_id, "MIGRATED", {"app_id": app_id})

    run = CutoverRun(
        run_id=f"cut-{app_id}-stub",
        app_id=app_id,
        source="real",
        steps=list(steps),
        history=history,
        result="MIGRATED",
        explanation="All gates passed at every step.",
    )
    store.save_cutover(run)
    return run

function layoutNodes(nodes, edges) {
  // Simple topological layout: bucket by longest-path depth
  const inDeg = {}
  const graph = {}
  nodes.forEach(n => { inDeg[n.id] = 0; graph[n.id] = [] })
  edges.forEach(e => {
    inDeg[e.target] = (inDeg[e.target] || 0) + 1
    graph[e.source] = [...(graph[e.source] || []), e.target]
  })
  // BFS by level
  const levels = {}
  const queue = nodes.filter(n => inDeg[n.id] === 0).map(n => n.id)
  queue.forEach(id => { levels[id] = 0 })
  while (queue.length) {
    const id = queue.shift()
    ;(graph[id] || []).forEach(nxt => {
      levels[nxt] = Math.max(levels[nxt] || 0, (levels[id] || 0) + 1)
      queue.push(nxt)
    })
  }
  const cols = {}
  nodes.forEach(n => {
    const lv = levels[n.id] || 0
    cols[lv] = (cols[lv] || 0)
    cols[lv]++
  })
  const colIdx = {}
  return nodes.map(n => {
    const lv = levels[n.id] || 0
    colIdx[lv] = (colIdx[lv] || 0)
    const pos = { x: lv * 220, y: colIdx[lv] * 100 }
    colIdx[lv]++
    return { ...n, position: pos }
  })
}

          setFlashRed(true)
          setTimeout(() => setFlashRed(false), 3000)
          if (ev.type === 'ROLLED_BACK') {
            setRolledBackMsg(ev.payload?.llm_explanation || ev.payload?.reason || 'Rolled back')
          }
          setMetrics(prev => ({
            error_rate: ev.payload?.error_rate ?? prev.error_rate,
            p95_ms: ev.payload?.p95_ms ?? prev.p95_ms,
          }))
        }

        if (ev.type === 'MIGRATED') {
          setMigrated(true)
        }
      } catch {}
    }

    return () => es.close()
  }, [selectedApp])

  const startCutover = async () => {
    if (!selectedApp) return
    setWeightData([])
    setRolledBackMsg(null)
    setMigrated(false)
    await axios.post(`/runs/cutover/${selectedApp}`, {}).catch(() => {})
    // Poll for cutover result
    let tries = 0
    const poll = setInterval(async () => {
      tries++
      try {
        const r = await axios.get(`/cutovers/${selectedApp}`)
        setCutover(r.data)
        if (['MIGRATED', 'ROLLED_BACK', 'ABORTED'].includes(r.data.result)) clearInterval(poll)
      } catch {}
      if (tries > 60) clearInterval(poll)
    }, 1000)
  }

  const toggleBadWave = (enabled) => {
    if (!selectedApp) return
    axios.post('/demo/bad-wave', { app_id: selectedApp, enabled }).catch(() => {})
  }

  const fixAndRetry = async () => {
    await axios.post('/demo/bad-wave', { app_id: selectedApp, enabled: false }).catch(() => {})
    setRolledBackMsg(null)
    setMigrated(false)
    setWeightData([])
    await startCutover()
  }

  const steps = cutover?.history || []
  const stepPassFail = STEPS.map(w => {
    const h = steps.find(s => s.weight === w)
    return { weight: w, gate: h?.gate }
  })

  return (
    <div className={`space-y-4 transition-colors ${flashRed ? 'bg-red-950/20' : ''} rounded-lg p-1`}>
      {/* App selector + controls */}
      <div className="flex gap-3 items-center flex-wrap">
        <select
          value={selectedApp}
          onChange={e => { setSelectedApp(e.target.value); setCutover(null); setWeightData([]); setRolledBackMsg(null); setMigrated(false) }}
          className="bg-gray-800 border border-gray-700 text-sm rounded px-3 py-2 text-gray-100"
        >
          {apps.map(a => <option key={a.app_id} value={a.app_id}>{a.app_id}</option>)}
        </select>

        <button
          onClick={startCutover}
          className="px-4 py-2 bg-teal-600 hover:bg-teal-500 text-white text-sm font-semibold rounded-lg"
        >
          Cut Over
        </button>

        <div className="flex items-center gap-2">
          <span className="text-xs text-gray-400">Bad wave:</span>
          <button onClick={() => toggleBadWave(true)} className="px-2 py-1 bg-red-700 hover:bg-red-600 text-white text-xs rounded">ON</button>
          <button onClick={() => toggleBadWave(false)} className="px-2 py-1 bg-gray-700 hover:bg-gray-600 text-white text-xs rounded">OFF</button>
        </div>

        {rolledBackMsg && (
          <button
            onClick={fixAndRetry}
            className="px-3 py-1.5 bg-amber-600 hover:bg-amber-500 text-white text-xs rounded-lg"
          >
            Fix &amp; Retry
          </button>
        )}
      </div>

      {/* Step badges */}
      <div className="flex gap-3">
        {stepPassFail.map(s => (
          <div
            key={s.weight}
            className={`px-4 py-2 rounded-lg text-sm font-mono border ${
              s.gate === 'PASS' ? 'bg-teal-900/30 border-teal-500/30 text-teal-300' :
              s.gate === 'FAIL' ? 'bg-red-900/30 border-red-500/30 text-red-300' :
              'bg-gray-800 border-gray-700 text-gray-500'
            }`}
          >
            {s.weight}% {s.gate === 'PASS' ? '✓' : s.gate === 'FAIL' ? '✗' : '—'}
          </div>
        ))}
      </div>

      {/* Metrics */}
      <div className="flex gap-6">
        {[
          { label: 'Error rate', value: metrics.error_rate != null ? `${(metrics.error_rate * 100).toFixed(1)}%` : '—', danger: metrics.error_rate > 0.02 },
          { label: 'p95 latency', value: metrics.p95_ms != null ? `${metrics.p95_ms.toFixed(0)} ms` : '—', danger: metrics.p95_ms > 800 },
        ].map(m => (
          <div key={m.label} className="bg-gray-900 rounded-lg border border-gray-800 px-4 py-3">
            <div className={`text-xl font-bold ${m.danger ? 'text-red-400' : 'text-gray-100'}`}>{m.value}</div>
            <div className="text-xs text-gray-500">{m.label}</div>
          </div>
        ))}
      </div>

      {/* Weight chart */}
      <div className="bg-gray-900 rounded-lg border border-gray-800 p-4 h-48">
        <h3 className="text-xs font-semibold text-gray-400 mb-2">Traffic Split (legacy / target)</h3>
        <ResponsiveContainer width="100%" height="85%">
          <AreaChart data={weightData} margin={{ top: 0, right: 0, left: -20, bottom: 0 }}>
            <CartesianGrid strokeDasharray="3 3" stroke="#374151" />
            <XAxis dataKey="ts" tick={{ fontSize: 9, fill: '#6b7280' }} interval="preserveStartEnd" />
            <YAxis domain={[0, 100]} tick={{ fontSize: 9, fill: '#6b7280' }} />
            <Tooltip contentStyle={{ background: '#111827', border: '1px solid #374151', fontSize: 11 }} />
            <Area type="monotone" dataKey="legacy" stackId="1" stroke="#ef4444" fill="#ef444440" name="Legacy" />
            <Area type="monotone" dataKey="target" stackId="1" stroke="#10b981" fill="#10b98140" name="Target" />
          </AreaChart>
        </ResponsiveContainer>
      </div>

      {/* Rollback alert */}
      {rolledBackMsg && !migrated && (
        <div className="bg-red-900/20 border border-red-500/30 rounded-lg p-4">
          <div className="text-red-400 font-semibold text-sm mb-1">⚠ Rolled Back</div>
          <div className="text-red-300 text-sm">{rolledBackMsg}</div>
        </div>
      )}

      {/* Migrated banner */}
      {migrated && (
        <div className="bg-teal-900/20 border border-teal-500/30 rounded-lg p-4 text-center">
          <div className="text-teal-400 font-bold text-lg">✓ Migrated</div>
          <div className="text-teal-300 text-sm mt-1">All gates passed. Traffic fully on target.</div>
        </div>
      )}
    </div>
  )
}

"""
data/generate_fleet.py
Generate a synthetic fleet of ~1,000 apps as fleet.json.

Usage:
    python data/generate_fleet.py --count 1000 --seed 42
"""
from __future__ import annotations

import argparse
import json
import random
import sys
from datetime import datetime, timezone
from pathlib import Path

try:
    import networkx as nx
    from faker import Faker
except ImportError:
    print("Missing dependencies. Run: pip install faker networkx")
    sys.exit(1)

# ---------------------------------------------------------------------------
# Constants
# ---------------------------------------------------------------------------

SERVICES = ["api", "service", "worker", "gateway", "processor", "hub", "engine", "platform"]
TEAMS = [
    "team-commerce", "team-infra", "team-data", "team-platform",
    "team-ml", "team-ops", "team-security", "team-frontend",
]
BUSINESS_UNITS = ["Retail", "Platform", "Data", "ML", "Finance", "Ops", "Security", "Growth"]
PORTS = [8080, 8443, 3000, 5000, 4000]

ALL_FINDINGS = [
    "SG_OPEN_SSH", "SG_OPEN_APP", "EBS_UNENCRYPTED", "IMDSV1",
    "OLD_AMI", "NO_VPC_SEGMENTATION", "PUBLIC_IP", "MISSING_TAGS",
]
GRAY_EXTRA = ["HARDCODED_IP"]

# Tier distribution targets
TIER_DIST = {"GOLDEN": 0.62, "GRAY": 0.28, "RED": 0.10}


# ---------------------------------------------------------------------------
# IP pool
# ---------------------------------------------------------------------------

def _ip_pool(count: int, rng: random.Random) -> list[str]:
    xs = list(range(2, 255))
    ys = list(range(2, 255))
    used: set[str] = set()
    ips: list[str] = []
    while len(ips) < count:
        x = rng.choice(xs)
        y = rng.choice(ys)
        ip = f"10.10.{x}.{y}"
        if ip not in used:
            used.add(ip)
            ips.append(ip)
    return ips


# ---------------------------------------------------------------------------
# App name generation
# ---------------------------------------------------------------------------

def _app_name(fake: Faker, rng: random.Random) -> str:
    adj = fake.word(ext_word_list=None)  # any word
    noun = fake.word(ext_word_list=None)
    service = rng.choice(SERVICES)
    # normalise: lowercase, only alphanum and hyphens
    parts = [p.lower().replace("_", "-") for p in [adj, noun, service]]
    parts = ["".join(c for c in p if c.isalnum() or c == "-").strip("-") for p in parts]
    parts = [p for p in parts if p]
    return "-".join(parts) if parts else f"app-{rng.randint(1000,9999)}"


# ---------------------------------------------------------------------------
# Dependency DAG
# ---------------------------------------------------------------------------

def _build_dag(app_ids: list[str], rng: random.Random, max_depth: int = 4, max_in_degree: int = 3) -> nx.DiGraph:
    g = nx.DiGraph()
    g.add_nodes_from(app_ids)
    for i, app_id in enumerate(app_ids[1:], start=1):
        # Each app can depend on 0–2 earlier apps (guarantees DAG)
        n_deps = rng.choices([0, 1, 2], weights=[0.45, 0.40, 0.15])[0]
        candidates = [a for a in app_ids[:i] if g.in_degree(a) < max_in_degree]
        if not candidates:
            continue
        # Prevent paths longer than max_depth
        chosen = rng.sample(candidates, min(n_deps, len(candidates)))
        for dep in chosen:
            g.add_edge(app_id, dep)  # consumer → provider
    return g


# ---------------------------------------------------------------------------
# Findings per tier
# ---------------------------------------------------------------------------

def _findings_for_tier(tier: str, rng: random.Random) -> list[str]:
    if tier == "GOLDEN":
        k = rng.randint(2, len(ALL_FINDINGS))
        return rng.sample(ALL_FINDINGS, k)
    elif tier == "GRAY":
        k = rng.randint(2, len(ALL_FINDINGS))
        base = rng.sample(ALL_FINDINGS, k)
        # GRAY always has HARDCODED_IP
        if "HARDCODED_IP" not in base:
            base.append("HARDCODED_IP")
        return base
    else:  # RED
        return list(ALL_FINDINGS) + ["HARDCODED_IP"]


# ---------------------------------------------------------------------------
# Main generator
# ---------------------------------------------------------------------------

def generate(count: int, seed: int) -> list[dict]:
    fake = Faker()
    Faker.seed(seed)
    rng = random.Random(seed)

    # Assign tiers
    tiers: list[str] = []
    for tier, frac in TIER_DIST.items():
        tiers.extend([tier] * round(count * frac))
    # Pad/trim to exact count
    while len(tiers) < count:
        tiers.append("GOLDEN")
    tiers = tiers[:count]
    rng.shuffle(tiers)

    # Generate unique app IDs
    app_ids: list[str] = []
    seen: set[str] = set()
    for i in range(count):
        for _ in range(20):
            name = _app_name(fake, rng)
            uid = f"syn-{i:05d}-{name}"[:40]
            if uid not in seen:
                seen.add(uid)
                app_ids.append(uid)
                break
        else:
            app_ids.append(f"syn-{i:05d}")

    ips = _ip_pool(count, rng)
    dag = _build_dag(app_ids, rng)

    apps: list[dict] = []
    for i, app_id in enumerate(app_ids):
        tier = tiers[i]
        team = rng.choice(TEAMS)
        bu = rng.choice(BUSINESS_UNITS)
        port = rng.choice(PORTS)
        private_ip = ips[i]
        findings = _findings_for_tier(tier, rng)
        stateful = tier == "RED"
        deps = list(dag.successors(app_id))  # providers
        ami_age = rng.randint(200, 1200)

        app = {
            "app_id": app_id,
            "source": "synthetic",
            "name": app_id.replace("syn-", "").replace("-", " ").title(),
            "owner": team,
            "business_unit": bu,
            "runtime": {
                "type": "ec2",
                "instance_ids": [f"i-{rng.randint(0,0xffffffffffff):012x}"],
                "ami_id": f"ami-{rng.randint(0,0xffffffff):08x}",
                "ami_age_days": ami_age,
                "instance_type": rng.choice(["t3.micro", "t3.small", "t3.medium", "m5.large"]),
                "port": port,
                "stateful": stateful,
            },
            "network": {
                "vpc_id": "vpc-legacy",
                "vpc_has_private_subnet": False,
                "subnet_public": True,
                "public_ip": True,
                "private_ip": private_ip,
                "sg_ingress": [
                    {"port": 22, "cidr": "0.0.0.0/0"},
                    {"port": port, "cidr": "0.0.0.0/0"},
                ],
            },
            "storage": {
                "ebs_encrypted": False,
                "volume_gb": rng.choice([8, 16, 32, 50]),
            },
            "metadata": {"imds_v2_required": False},
            "config": {
                "env": {
                    "UPSTREAM_URL": f"http://{ips[rng.randint(0, count-1)]}:{port}"
                } if deps else {}
            },
            "depends_on": deps[:3],  # max 3 listed
            "tags": {
                "Name": app_id,
                "app": app_id,
                "team": team,
                "depends-on": deps[0] if deps else "",
            },
            "findings": findings,
            "status": "DISCOVERED",
        }
        apps.append(app)

    return apps


# ---------------------------------------------------------------------------
# Entry point
# ---------------------------------------------------------------------------

def main():
    parser = argparse.ArgumentParser(description="Synthetic fleet generator")
    parser.add_argument("--count", type=int, default=1000)
    parser.add_argument("--seed", type=int, default=42)
    parser.add_argument("--out", type=str, default="data/fleet.json")
    args = parser.parse_args()

    print(f"Generating {args.count} synthetic apps (seed={args.seed})...")
    apps = generate(args.count, args.seed)

    out_path = Path(args.out)
    out_path.parent.mkdir(parents=True, exist_ok=True)
    out_path.write_text(json.dumps(apps, indent=2))

    # Stats
    tiers = {"GOLDEN": 0, "GRAY": 0, "RED": 0}
    finding_counts: dict[str, int] = {}
    has_deps = 0
    for a in apps:
        # We need to determine tier from findings
        if "HARDCODED_IP" in a["findings"] or len(a["depends_on"]) > 2:
            tiers["GRAY"] += 1
        elif a["runtime"]["stateful"]:
            tiers["RED"] += 1
        else:
            tiers["GOLDEN"] += 1
        if a["depends_on"]:
            has_deps += 1
        for f in a["findings"]:
            finding_counts[f] = finding_counts.get(f, 0) + 1

    print(f"\n=== Fleet stats ===")
    print(f"  Total apps  : {len(apps)}")
    print(f"  GOLDEN      : {tiers['GOLDEN']}")
    print(f"  GRAY        : {tiers['GRAY']}")
    print(f"  RED         : {tiers['RED']}")
    print(f"  With deps   : {has_deps}")
    print(f"\n  Top findings:")
    for f, c in sorted(finding_counts.items(), key=lambda x: -x[1])[:5]:
        print(f"    {f}: {c}")
    print(f"\nWritten to {out_path}")


if __name__ == "__main__":
    main()
